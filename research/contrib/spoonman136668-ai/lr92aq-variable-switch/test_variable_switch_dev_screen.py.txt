"""LR-92AQ explicit research controls, only unprotected dev seeds."""
import json
import math
import unittest
from dataclasses import fields
from pathlib import Path

from lr92aq_variable_switch_dev import (
    DEVELOPMENT_SEEDS, FAMILIES, WORLDS_PER_SEED_FAMILY,
    PublicObservation, development_world, fork_continuing_B, score_public_world,
    MIN_DWELL, MAX_DWELL, EPISODE_LENGTH,
)
from public_dual_timescale_cells import PublicDualTimescaleCells
from variable_switch_dev_screen import (
    MODEL_TYPES, PRIMARY_SEEDS_FORBIDDEN, METRICS, prequential,
    evaluate, decision, summarized, learned_bytes,
)


class VariableSwitchComparisonTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.rows = evaluate()

    def test_exact_dev_scope_no_protected_use(self):
        self.assertEqual(tuple(DEVELOPMENT_SEEDS),(73004303,73004304))
        self.assertEqual(tuple(FAMILIES),("xor","and","random14"))
        self.assertEqual(WORLDS_PER_SEED_FAMILY,4)
        self.assertTrue(PRIMARY_SEEDS_FORBIDDEN.isdisjoint(DEVELOPMENT_SEEDS))
        for seed in PRIMARY_SEEDS_FORBIDDEN:
            with self.assertRaises(ValueError):development_world(seed,"xor",0)
        self.assertEqual(len(self.rows),24)
        self.assertEqual({(r["seed"],r["family"],r["index"])
                          for r in self.rows},
                         {(s,f,i) for s in DEVELOPMENT_SEEDS for f in FAMILIES
                          for i in range(4)})

    def test_public_inference_surface_has_no_latent_labels(self):
        self.assertEqual(tuple(x.name for x in fields(PublicObservation)),
                         ("index","episode_start","action","outcome"))
        for row in self.rows:
            world=development_world(row["seed"],row["family"],row["index"])
            self.assertEqual(world.public_tape_hash(),row["public_sha256"])
            for public in world.public:
                self.assertNotIn("phase",public.__dict__)
                self.assertNotIn("dwell_lengths",public.__dict__)
                self.assertNotIn("family",public.__dict__)

    def test_exact_memory_budgets_and_checkpoints(self):
        self.assertEqual(set(MODEL_TYPES),{
            "YPT4_candidate","FullHistory14","FixedWindow14",
            "HardMAP_HMM","Bounded_BOCPD896"})
        expect={"YPT4_candidate":928,"FullHistory14":480,
                "FixedWindow14":832,"HardMAP_HMM":928,
                "Bounded_BOCPD896":896}
        for name,cls in MODEL_TYPES.items():
            self.assertEqual(learned_bytes(cls()),expect[name])
            m=cls()
            self.assertEqual(len(m.snapshot()),expect[name]+36)
            self.assertEqual(cls.restore(m.snapshot()).snapshot(),m.snapshot())
        for row in self.rows:
            for name,out in row["models"].items():
                self.assertEqual(out["bytes"],expect[name])
                self.assertEqual(out["checkpoint_bytes"],expect[name]+36)

    def test_full_prequential_update_count_and_finite_scores(self):
        for row in self.rows:
            for name,out in row["models"].items():
                self.assertEqual(out["updates"],row["public_events"])
                self.assertEqual(out["predicts"],row["public_events"])
                self.assertEqual(set(out["score"]),set(METRICS))
                self.assertTrue(all(0<=v<=1 and math.isfinite(v) for v in
                                    out["score"].values()),name)
                self.assertEqual(len(out["snapshot_sha256"]),64)

    def test_frozen_preprimary_source_has_distinct_stream_contract(self):
        # Existing dev fixture CANNOT be relabeled as the new primary.
        import json
        doc=json.loads(Path(__file__).with_name("prereg.json.txt").read_text())
        self.assertEqual(doc["status"],
          "PROTOCOL_FROZEN_SOURCE_NOT_BORN_PRIMARY_UNRUN_OWNER_UNVERIFIED")
        self.assertEqual((MIN_DWELL,MAX_DWELL,EPISODE_LENGTH),(48,88,7))
        self.assertEqual(doc["prospective_design"]["observed_public_events_per_world"],192)
        self.assertEqual(doc["prospective_design"]["phase_lengths"]["B"]["max"],80)
        lengths={r["public_events"] for r in self.rows}
        self.assertNotEqual(lengths,{192})
        self.assertFalse(doc["authority"]["official_scientific_first_use_authorized"])
        self.assertFalse(doc["gates_before_primary"]["deterministic_generator_source_born"])

    def test_compare_same_world_and_reproducible_without_timing(self):
        row=self.rows[0]
        world=development_world(row["seed"],row["family"],row["index"])
        direct=score_public_world(world,PublicDualTimescaleCells())
        self.assertAlmostEqual(
            direct["A_return_early16"],
            row["models"]["YPT4_candidate"]["score"]["return_1_16"])
        again=prequential(world,PublicDualTimescaleCells)
        self.assertEqual(again["score"],row["models"]["YPT4_candidate"]["score"])
        self.assertEqual(again["snapshot_sha256"],
                         row["models"]["YPT4_candidate"]["snapshot_sha256"])

    def test_counterfactual_prefix_is_identical_but_future_labels_differ(self):
        world=development_world(DEVELOPMENT_SEEDS[0],"random14",1)
        fork=fork_continuing_B(world)
        switch=sum(world.evaluator_only.dwell_lengths[:2])
        self.assertEqual(world.public[:switch],fork.public[:switch])
        self.assertEqual(world.public[switch].action,fork.public[switch].action)
        self.assertEqual(world.public[switch].index,fork.public[switch].index)
        self.assertEqual(world.public[switch].episode_start,
                         fork.public[switch].episode_start)
        # First return outcome is necessarily opposite in the counterfactual
        # for identical noise draw. The forecast cannot use that outcome.
        self.assertNotEqual(world.public[switch].outcome,
                            fork.public[switch].outcome)

    def test_all_six_seed_family_groups_reported_fail_closed(self):
        d=decision(self.rows)
        self.assertEqual(len(d["groups"]),6)
        self.assertIn(d["classification"],("DEV_NO_GO","DEV_PLAUSIBLE_UNCONFIRMED"))
        for g in d["groups"]:
            self.assertIn(g["best_return_control"],MODEL_TYPES)
            self.assertNotEqual(g["best_return_control"],"YPT4_candidate")
            self.assertTrue(math.isfinite(g["return_5_16_gain"]))
            self.assertTrue(math.isfinite(g["B_regression"]))
            self.assertEqual(g["screening_pass"],
                             g["return_5_16_gain"]>=.01 and g["B_regression"]<=.01)

    def test_summary_does_not_invent_native_training_or_primary(self):
        d=summarized(self.rows)
        self.assertFalse(d["proposed_primary_executed"])
        self.assertFalse(d["canonical_owner_checkpoint_used"])
        self.assertFalse(d["actual_native_yggdrasil_training"])
        self.assertEqual(d["worlds"],24)
        self.assertEqual(d["per_model_scores_unweighted_world_means"].keys(),
                         MODEL_TYPES.keys())
        self.assertEqual(sum(r["public_events"] for r in self.rows),d["public_events"])

if __name__=="__main__":
    unittest.main()
