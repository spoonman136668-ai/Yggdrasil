"""LR-92BB disjoint source-free two-stage causal worlds with UNANNOUNCED A/B/A.

The learner receives only (s1,x1,s2,x2,y) and one constant view address.
No public regime tag, true function identity, phase clock, or future events.
"""
from __future__ import annotations
from dataclasses import dataclass
import hashlib
import random
from mechanism import FEATURE_NAMES, FEATURES

SEEDS=(94771101,94771102)
FAMILIES=('cubic_parent_rewrite','cubic_disjoint_parent','quartic_unfamiliar')
REPS=4
FORBIDDEN=(73004303,73004304,74009171,74009172,75011801,75011802,
           76021921,76021922,77033101,77033102,88041001,88041002,
           89051001,89051002,90161101,90161102,90161103,
           92111911,92111912,92111913,93490111,93490112,
           929209101,929209102,929209103)
FLIP_TERMINAL=.07
FLIP_TRANSITION=.25

def rng(seed,family,rep,tag):
    b=hashlib.sha256(f'LR92BB|{seed}|{family}|{rep}|{tag}'.encode()).digest()
    return random.Random(int.from_bytes(b[:8],'big'))

def true_law(family,rep,regime,ctx):
    f=FEATURES[ctx]
    if family=='cubic_parent_rewrite':
        terms=(12,3,2) if regime==0 else (13,2,0)
    elif family=='cubic_disjoint_parent':
        terms=(11,4,0) if regime==0 else (14,1,0)
    elif family=='quartic_unfamiliar':
        terms=(0,5) if regime==0 else (7,8,0)
    else:raise ValueError('FAMILY')
    y=(sum(f[i] for i in terms) + (rep&1))&1
    if family=='quartic_unfamiliar' and regime==0:
        y^=f[1]*f[2]*f[3]*f[4]
    return y

@dataclass(frozen=True)
class Event:
    identifier:str
    s1:int
    x1:int
    s2:int
    x2:int
    y:int
    def ctx(self):return (self.s1<<3)|(self.x1<<2)|(self.s2<<1)|self.x2
    def views(self):return ((self.s1,0),(self.s1,1))
    def row(self):return {'s1':self.s1,'x1':self.x1,'s2':self.s2,'x2':self.x2,'y':self.y}

@dataclass(frozen=True)
class World:
    seed:int
    family:str
    rep:int
    dwell:tuple[int,int,int]
    events:tuple[Event,...]
    public_sha256:str


def world(seed,family,rep):
    if seed not in SEEDS or family not in FAMILIES or type(rep)!=int or not 0<=rep<REPS:raise ValueError('WORLD_SCOPE')
    z=rng(seed,family,rep,'dwell');dwell=(z.randrange(74,96),z.randrange(70,90),z.randrange(53,75))
    actions=rng(seed,family,rep,'public_action'); n1=rng(seed,family,rep,'transition_noise');ny=rng(seed,family,rep,'terminal_noise')
    events=[]
    for i in range(sum(dwell)):
        # The phase exists only inside evaluator's generator; NEVER inside Event.
        phase=0 if i<dwell[0] or i>=dwell[0]+dwell[1] else 1
        s1=actions.randrange(2);x1=actions.randrange(2);x2=actions.randrange(2)
        s2=(s1^x1)^int(n1.random()<FLIP_TRANSITION)
        ctx=s1*8+x1*4+s2*2+x2
        y=true_law(family,rep,phase,ctx)^int(ny.random()<FLIP_TERMINAL)
        ident=hashlib.sha256(f'LR92BB|{seed}|{family}|{rep}|{i}'.encode()).hexdigest()
        events.append(Event(ident,s1,x1,s2,x2,y))
    raw=bytes(b for e in events for b in (e.s1,e.x1,e.s2,e.x2,e.y))
    return World(seed,family,rep,dwell,tuple(events),hashlib.sha256(raw).hexdigest())

def worlds():return tuple(world(s,f,r) for s in SEEDS for f in FAMILIES for r in range(REPS))
