"""LR-92BA independent read-only reproduction of frozen learner Brier outcomes.

Does NOT import the candidate, compare harness, Yggdrasil native model, or owner
state. Regenerates public synthetic fixtures only and separately implements
all seven fixed learning policies from the preregistered equations.
"""
from __future__ import annotations
import hashlib,json,math
from pathlib import Path
from worlds import world,SEEDS,FAMILIES,REPS,worlds

SOURCE_LOCK='6f7eaea55676ad3c0e5a4d679a226b00292b1061'
ARMS=('repair_basis','exact_symbolic_bayes','beta_full','beta_window32','beta_decay24','fragile_contradiction','no_learning')
CONTROLS=('exact_symbolic_bayes','beta_full','beta_window32','beta_decay24')
NOISE=.12
ODDS=math.log((1-NOISE)/NOISE)


def bits(stage,addr):
    if stage=='s':
        s1=(addr>>1)&1;x1=addr&1
        return (1,s1,x1,s1*x1)
    if stage=='y':
        s1=(addr>>3)&1;x1=(addr>>2)&1;s2=(addr>>1)&1;x2=addr&1
        return (1,s1,x1,s2,x2,s1*x1,s1*s2,s1*x2,x1*s2,x1*x2,s2*x2)
    raise ValueError('stage')

HYP={s:[tuple((m&sum(int(x)<<i for i,x in enumerate(bits(s,k)))).bit_count()%2 for k in range(n))
        for m in range(1<<nf)] for s,n,nf in [('s',4,4),('y',16,11)]}


def address(s,v,s1,x1,s2,x2):
    return (s1*2+x1) if s=='s' else (s1*8+x1*4+s2*2+x2)


class AuditLearner:
    def __init__(self,arm):
        self.arm=arm
        self.n=0
        self.obs=[]
        self.counts={(s,v):[[0,0] for _ in range(n)] for s,n in [('s',4),('y',16)] for v in (0,1)}
        self.scores={k:[(-i.bit_count() if arm=='repair_basis' else -.4*i.bit_count()) for i in range(len(HYP[k[0]]))] for k in self.counts}
        self.invalid=set();self.first={}
    def prediction(self,s,v,s1,x1,s2,x2):
        arm=self.arm;key=s,v;addr=address(s,v,s1,x1,s2,x2)
        if arm=='no_learning':return .5
        if arm=='fragile_contradiction':
            return .5 if key in self.invalid or (key,addr) not in self.first else float(self.first[key,addr])
        if arm in ('repair_basis','exact_symbolic_bayes'):
            scores=self.scores[key];hypotheses=HYP[s]
            if arm=='repair_basis':
                winner=max(range(len(scores)),key=scores.__getitem__)
                n=sum(a+b for a,b in self.counts[key])
                bad=sum(cell[1-pred] for cell,pred in zip(self.counts[key],hypotheses[winner]))
                if n<4 or n>=24 and bad/n>.26:return .5
                return .12+.76*hypotheses[winner][addr]
            peak=max(scores);norm=1e-99;positive=0.
            for mask,score in enumerate(scores):
                w=math.exp(score-peak);norm+=w
                positive+=w*hypotheses[mask][addr]
            return .12+.76*positive/norm
        q=('s',v,s1,x1) if s=='s' else ('y',v,s1,x1,s2,x2)
        obs=self.obs[-64:] if arm=='beta_window32' else self.obs
        n=k=0.
        for i,(_,address_key,truth) in enumerate(obs):
            if address_key!=q:continue
            weight=(2**(-(len(obs)-1-i)/48)) if arm=='beta_decay24' else 1.
            n+=weight;k+=weight*truth
        return (1+k)/(2+n)
    def update(self,v,s1,x1,s2,x2,y):
        self.n+=1
        for s,truth in [('s',s2),('y',y)]:
            key=s,v;addr=address(s,v,s1,x1,s2,x2)
            if self.arm in ('beta_full','beta_window32','beta_decay24'):
                q=('s',v,s1,x1) if s=='s' else ('y',v,s1,x1,s2,x2)
                self.obs.append((self.n,q,truth))
            elif self.arm=='fragile_contradiction':
                if key not in self.invalid:
                    k=key,addr
                    if k in self.first and self.first[k]!=truth:self.invalid.add(key)
                    else:self.first[k]=truth
            elif self.arm in ('repair_basis','exact_symbolic_bayes'):
                self.counts[key][addr][truth]+=1
                mag=8 if self.arm=='repair_basis' else ODDS
                for m,pred in enumerate(HYP[s]):
                    self.scores[key][m]+=mag if pred[addr]==truth else -mag


def independently_score(world,arm):
    model=AuditLearner(arm)
    a,b,c=world.dwell;rows={'A':[],'B':[],'return':[]}
    for i,e in enumerate(world.events):
        phase='A' if i<a else 'B' if i<a+b else 'return'
        ps=model.prediction('s',e.v,e.s1,e.x1,e.s2,e.x2)
        py=model.prediction('y',e.v,e.s1,e.x1,e.s2,e.x2)
        rows[phase].append(((ps-e.s2)**2,(py-e.y)**2))
        model.update(*e.public())
    avg=lambda rr,k:sum(z[k] for z in rr)/len(rr)
    return {'A_terminal':avg(rows['A'],1),'B_terminal':avg(rows['B'],1),
            'B_transition':avg(rows['B'],0),
            'return16_terminal':avg(rows['return'][:16],1),
            'return16_transition':avg(rows['return'][:16],0)}


def audit(result_path,protocol_path='frozen_dev_protocol.json',manifest_path='development_manifest.json'):
    p=Path(protocol_path);m=Path(manifest_path);raw=Path(result_path).read_bytes()
    data=json.loads(raw)
    protocol=json.loads(p.read_bytes());manifest=json.loads(m.read_bytes())
    assert protocol['state']=='FROZEN_BEFORE_SCORING'
    assert protocol['data']['manifest_sha256']==hashlib.sha256(m.read_bytes()).hexdigest()
    for name,expected in protocol['source_sha256'].items():
        assert hashlib.sha256(Path(name).read_bytes()).hexdigest()==expected,('SOURCE_DRIFT',name)
    assert data['schema']=='lr92ba.noisy_causal_revision.dev.v1' and data['worlds']==24
    assert data['episodes_per_arm']==protocol['data']['episodes_per_arm']==3738
    assert data['arms']==list(ARMS) and len(data['rows'])==len(manifest)==24
    assert data['protected_first_use'] is False and data['native_model_executed'] is False
    assert data['canonical_checkpoint_changed'] is False
    checked=0;retained=0;flips=0;rows=data['rows']
    for rec,spec in zip(rows,manifest):
        key=(rec['seed'],rec['family'],rec['rep'])
        assert key==(spec['seed'],spec['family'],spec['rep'])
        w=world(*key)
        assert rec['public_sha256']==spec['public_sha256']==w.public_sha256
        assert list(w.dwell)==spec['dwell']==rec['dwell']
        assert rec['events']==spec['events']==len(w.events)
        assert list(w.flip_counts)==rec['flips']==spec['flip_counts']
        assert set(rec['models'])==set(ARMS)
        flips+=sum(w.flip_counts)
        for arm in ARMS:
            saved=rec['models'][arm]
            assert saved['updates']==len(w.events)
            assert len(saved['checkpoint_sha256'])==64
            assert saved['checkpoint_bytes']>0 and saved['elapsed_seconds']>=0
            independent=independently_score(w,arm)
            for field,v in independent.items():
                assert abs(v-saved['scores'][field])<1e-10,('INDEPENDENT_SCORE_MISMATCH',key,arm,field,v,saved['scores'][field])
            checked+=1
        repair=rec['models']['repair_basis']
        assert repair['checkpoint_bytes']==236
        assert set(repair['cold_sha'])=={'pre_B','pre_return'}
        assert all(len(v)==64 for v in repair['cold_sha'].values())
        before=repair['probes']['pre_B']['terminal'];after=repair['probes']['pre_return']['terminal']
        assert before==after,('COLD_RETENTION_FAIL',key,before,after)
        retained+=1
    assert checked==24*len(ARMS) and retained==24
    summary={str(seed):{arm:{field:sum(r['models'][arm]['scores'][field] for r in rows if r['seed']==seed)/12
                         for field in ('A_terminal','B_terminal','B_transition','return16_terminal')}
                      for arm in ARMS} for seed in SEEDS}
    flags={}
    for seed in SEEDS:
        expected=summary[str(seed)];actual=data['seed_summary'][str(seed)]
        for arm in ARMS:
            for metric in expected[arm]:assert abs(expected[arm][metric]-actual[arm][metric])<1e-10
        b=min(expected[x]['B_terminal'] for x in CONTROLS)
        ret=min(expected[x]['return16_terminal'] for x in CONTROLS)
        c=expected['repair_basis']
        flags[str(seed)]=c['return16_terminal']<=ret-.01 and c['B_terminal']<=b+.01
    assert flags==data['seed_go']
    status='DEVELOPMENT_GO' if all(flags.values()) else 'DEVELOPMENT_NO_GO'
    assert status==data['decision']
    return {'schema':'lr92ba.independent_audit.v1','audit':'PASS','source_lock':SOURCE_LOCK,
            'worlds':24,'public_episodes_per_arm':3738,'arms':len(ARMS),'independently_reproduced_arm_world_scores':checked,
            'cold_retention_worlds':retained,'checkpoint_bytes':236,'flip_observations':flips,
            'seed_go':flags,'decision':status,'protected_first_use':False,'owner_state_changed':False,
            'raw_result_sha256':hashlib.sha256(raw).hexdigest()}

if __name__=='__main__':
    import sys
    if len(sys.argv)!=2:raise SystemExit('usage: audit_saved_results.py development_result.json')
    print(json.dumps(audit(sys.argv[1]),sort_keys=True))
